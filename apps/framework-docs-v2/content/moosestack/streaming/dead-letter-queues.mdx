# Dead letter queues

Use `DeadLetterQueue<T>` to capture records of type `T` that fail during stream processing. A dead letter queue is itself a stream of wrapper records: Moose keeps the original payload and adds error details about why that record failed.

## Create a dead letter queue

<LanguageTabs>
  <LanguageTabContent value="typescript">
```ts filename="app/dead-letter-queue.ts" copy
import { DeadLetterQueue } from "@514labs/moose-lib";

interface UserEvent {
  userId: string;
  action: string;
  timestamp: Date;
}

export const userEventDLQ = new DeadLetterQueue<UserEvent>("user_event_dlq");
```
  </LanguageTabContent>
  <LanguageTabContent value="python">
```py filename="app/dead_letter_queue.py" copy
from moose_lib import DeadLetterQueue

from pydantic import BaseModel
from datetime import datetime


class Record(BaseModel):
    id: str
    name: str
    timestamp: datetime

record_dlq = DeadLetterQueue[Record]("record_dlq")
```
  </LanguageTabContent>
</LanguageTabs>

### Attach to a transform

<LanguageTabs>
  <LanguageTabContent value="typescript">
```ts filename="app/transform-with-dlq.ts" copy
import { DeadLetterQueue, Stream } from "@514labs/moose-lib";

interface RawRecord {
  id: string;
  name: string;
  timestamp: Date;
}

interface ProcessedRecord extends RawRecord {
  processed: Date;
}

export const rawRecords = new Stream<RawRecord>("raw_records");
export const processedRecords = new Stream<ProcessedRecord>("processed_records");
export const recordDLQ = new DeadLetterQueue<RawRecord>("record_dlq");

rawRecords.addTransform(
  processedRecords,
  (record) => {
    if (!record.id) {
      throw new Error("Invalid id");
    }

    return {
      id: record.id,
      name: record.name,
      timestamp: record.timestamp,
      processed: new Date(),
    };
  },
  {
    deadLetterQueue: recordDLQ,
  }
);
```
  </LanguageTabContent>
  <LanguageTabContent value="python">
```py filename="app/transform_with_dlq.py" copy
from moose_lib import DeadLetterQueue, Stream, TransformConfig
from pydantic import BaseModel
from datetime import datetime

class RawRecord(BaseModel):
    id: str
    name: str
    timestamp: datetime

class ProcessedRecord(BaseModel):
    id: str
    name: str
    processed: datetime

raw_records = Stream[RawRecord]("raw_records")
processed_records = Stream[ProcessedRecord]("processed_records")
record_dlq = DeadLetterQueue[RawRecord]("record_dlq")

def process_record(record: RawRecord) -> ProcessedRecord:
    if not record.id:
        raise ValueError("Invalid id")

    return ProcessedRecord(
        id=record.id,
        name=record.name,
        processed=datetime.now(),
    )

raw_records.add_transform(
    destination=processed_records,
    transformation=process_record,
    config=TransformConfig(dead_letter_queue=record_dlq),
)
```
  </LanguageTabContent>
</LanguageTabs>


### Attach to a consumer

<LanguageTabs>
  <LanguageTabContent value="typescript">
```ts filename="app/consumer-with-dlq.ts" copy
import { ConsumerConfig, DeadLetterQueue, Stream } from "@514labs/moose-lib";

interface RawEvent {
  userId: string;
  action: string;
}

export const rawEvents = new Stream<RawEvent>("raw_events");
export const eventDLQ = new DeadLetterQueue<RawEvent>("event_dlq");

rawEvents.addConsumer(
  (event) => {
    if (event.action === "forbidden_action") {
      throw new Error("Forbidden action detected");
    }

    console.log(`Processing ${event.action} for ${event.userId}`);
  },
  {
    deadLetterQueue: eventDLQ,
  } satisfies ConsumerConfig<RawEvent>
);
```
  </LanguageTabContent>
  <LanguageTabContent value="python">
```py filename="app/consumer_with_dlq.py" copy
from moose_lib import ConsumerConfig, DeadLetterQueue, Stream
from pydantic import BaseModel

class RawEvent(BaseModel):
    user_id: str
    action: str

raw_events = Stream[RawEvent]("raw_events")
event_dlq = DeadLetterQueue[RawEvent]("event_dlq")

def process_event_consumer(event: RawEvent) -> None:
    if event.action == "forbidden_action":
        raise ValueError("Forbidden action detected")

    print(f"Processing {event.action} for {event.user_id}")

raw_events.add_consumer(
    consumer=process_event_consumer,
    config=ConsumerConfig(dead_letter_queue=event_dlq),
)
```
  </LanguageTabContent>
</LanguageTabs>

### Attach to an Ingest API

<LanguageTabs>
  <LanguageTabContent value="typescript">

import { DeadLetterQueue, IngestApi, Stream } from "@514labs/moose-lib";

interface UserEvent {
  id: string;
  userId: string;
  timestamp: Date;
  eventType: string;

const userEventDLQ = new DeadLetterQueue<UserEvent>("user_events_dlq");

from moose_lib import DeadLetterQueue, IngestApi, IngestConfigWithDestination, Stream
from pydantic import BaseModel

class UserEvent(BaseModel):
    id: str
    user_id: str
    timestamp: datetime
    event_type: str

user_events = Stream[UserEvent]("user_events")
user_event_dlq = DeadLetterQueue[UserEvent]("user_events_dlq")

user_events_api = IngestApi[UserEvent]("user-events", IngestConfigWithDestination(
    destination=user_events,
    dead_letter_queue=user_event_dlq,

## Dead letter payload format

When Moose writes a failed record into a dead letter queue, it wraps the original payload with a dead letter model that includes failure metadata alongside the original record. TypeScript exposes a base `DeadLetterModel` plus a typed `DeadLetter<T>` helper, while Python exposes `DeadLetterModel[T]` directly.

<LanguageTabs>
  <LanguageTabContent value="typescript">
```ts filename="app/dead-letter-types.ts" copy
interface DeadLetterModel {
  originalRecord: Record<string, any>;
  errorMessage: string;
  errorType: string;
  failedAt: Date;
  source: "api" | "transform" | "table";
}

interface DeadLetter<T> extends DeadLetterModel {
  asTyped: () => T;
}
```

  <thead>
    <tr>
      <th>Member</th>
      <th>Meaning</th>
    </tr>
  </thead>
  <tbody>
    <tr>
      <td><code>originalRecord</code></td>
      <td>The original failed payload before Moose wrapped it as a dead letter.</td>
    </tr>
    <tr>
      <td><code>errorMessage</code></td>
      <td>The error message Moose recorded for the failure.</td>
    </tr>
    <tr>
      <td><code>errorType</code></td>
      <td>The category of failure, such as a validation or transform error.</td>
    </tr>
    <tr>
      <td><code>failedAt</code></td>
      <td>When Moose wrote the record to the dead letter queue.</td>
    </tr>
    <tr>
      <td><code>source</code></td>
      <td>Where the failure happened, such as <code>api</code>, <code>transform</code>, or <code>table</code>.</td>
    </tr>
    <tr>
      <td><code>asTyped()</code></td>
      <td>Returns the original payload as the expected application type <code>T</code>. This helper is available on <code>DeadLetter&lt;T&gt;</code>.</td>
    </tr>
  </tbody>
</table>
  </LanguageTabContent>

from typing import Any, Generic, Literal, TypeVar

from pydantic import AliasGenerator, BaseModel, ConfigDict
from pydantic.alias_generators import to_camel

T = TypeVar("T")

class DeadLetterModel(BaseModel, Generic[T]):
    model_config = ConfigDict(
        alias_generator=AliasGenerator(
            serialization_alias=to_camel,
        )
    )

    original_record: Any
    error_message: str
    error_type: str
    failed_at: datetime.datetime
    source: Literal["api", "transform", "table"]

    def as_typed(self) -> T:
        ...

## Inspect or recover dead letters

Because a dead letter queue is itself a stream, you can attach consumers or transforms to it for monitoring or recovery.

<LanguageTabs>
  <LanguageTabContent value="typescript">
```ts filename="app/dlq-recovery.ts" copy
import { DeadLetterQueue } from "@514labs/moose-lib";

interface UserEvent {
  userId: string;
  action: string;
}

export const eventDLQ = new DeadLetterQueue<UserEvent>("event_dlq");

eventDLQ.addConsumer((deadLetter) => {
  console.error(deadLetter.errorMessage);
  const originalEvent = deadLetter.asTyped();
  console.log(originalEvent.userId);
});
```
  </LanguageTabContent>
  <LanguageTabContent value="python">
```py filename="app/dlq_recovery.py" copy
from moose_lib import DeadLetterQueue
from pydantic import BaseModel

class UserEvent(BaseModel):
    user_id: str
    action: str

event_dlq = DeadLetterQueue[UserEvent]("event_dlq")

def monitor_dead_letters(dead_letter) -> None:
    print(dead_letter.error_message)
    original_event = dead_letter.as_typed()
    print(original_event.user_id)

event_dlq.add_consumer(monitor_dead_letters)
```
  </LanguageTabContent>
</LanguageTabs>

<Callout type="info" title="Dead letter queues are for recovery, not silent drops">
Use a dead letter queue when failures should be retained for inspection or replay instead of disappearing from the stream pipeline.
</Callout>
